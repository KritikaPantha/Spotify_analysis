import sys
from awsglue.transforms import *
from awsglue.utils import getResolvedOptions
from pyspark.context import SparkContext
from awsglue.context import GlueContext
from awsglue.job import Job

# Get job parameters
args = getResolvedOptions(sys.argv, ['JOB_NAME'], 'S3_INPUT_PATH1', 'S3_INPUT_PATH2', 'S3_INPUT_PATH3', 'S3_OUTPUT_PATH')

# Initialize contexts
sc = SparkContext()
glue_context = GlueContext(sc)
spark = glue_context.spark_session
job = Job(glue_context)
job.init(args['JOB_NAME'], args)

#Read csv data from s3 using PySpark DataFrame
input_path1 = args['S3_INPUT_PATH1']
input_path2 = args['S3_INPUT_PATH2']
input_path3 = args['S3_INPUT_PATH3']
output_path = args['S3_OUTPUT_PATH']

#load the csv data into pyspark dataframe
albums_df = spark.read.option("header","true").csv("input_path1")
artists_df = spark.read.option("header","true").csv("input_path2")
track_df = spark.read.option("header","true").csv("input_path3")

# Join albums with artists on artist_id = id
albums_artists_df = albums_df.join(
    artists_df,
    albums_df["artist_id"] == artists_df["id"],"inner"
)

# Join the result with tracks on album_id = album_id
joined_df = albums_artists_df.join(
    track_df,
    albums_artists_df["track_id"] == track_df["track_id"], "inner"
)

# Select only id and track id
final_df = joined_df.select(artists_df["id"], track_df["track_id"])

# Write result to S3
final_df.write.mode("overwrite").option("header", "true").option("compression", "snappy").csv(output_path)

# Commit the job
job.commit()
